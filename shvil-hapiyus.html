שְׁבִיל הַפִּיּוּס
שֶׁל הַבַּעַל שֵׁם טוֹב הַקָּדוֹשׁ
✦
יְצִיאַת מִצְרַיִם לְנַפְשִׁי
בָּרִיא הוּא אָדָם אוֹהֵב: אֶת עַצְמוֹ, אֶת אָבִיו וְאֶת אִמּוֹ וְאֶת אֶחָיו, אֶת בֶּן/בַּת הַזּוּג וְאֶת יְלָדָיו, אֶת מִשְׁפַּחְתּוֹ וְאֶת בֵּיתוֹ, אֶת שְׁכֵנָיו וּשְׁכוּנָתוֹ, אֶת עַמּוֹ וְאֶת אַרְצוֹ, אֶת הַבּוֹרֵא וְאֶת הַבְּרִיאָה וְאֶת הַבְּרִיּוֹת.

"אַף אֶחָד לֹא חַיָּב לִי כְּלוּם. לֹא אַבָּא וְלֹא אִמָּא שֶׁלִּי, לֹא הָאַחִים וְהַאֲחָיוֹת, לֹא הַחֲבֵרִים וְלֹא הַחֶבְרָה, לֹא הַבּוֹרֵא, לֹא הַבְּרִיאָה, לֹא כָּל הַנִּבְרָאִים — אַף אֶחָד בְּכָל הַמְּצִיאוּת לֹא חַיָּב לִי כְּלוּם. וְלָכֵן אֲנִי בֶּן חוֹרִין, מְשֻׁחְרָר וְחָפְשִׁי לְהָאִיר לָהֶם."

מַשִּׂימַת הַפִּיּוּס
הַמְּשִׂימָה שֶׁלָּנוּ עַל מְנָת לִהְיוֹת בְּרִיאִים בְּנַפְשֵׁנוּ: הַפִּיּוּס — לִהְיוֹת מְפֻיָּסִים עִם הַחַיִּים בְּלִי מִשְׁקָעִים כְּלָל. כָּל עוֹד אֵינֶנּוּ מַגִּיעִים לְפִיּוּס עִם חֲוָיוֹת הֶעָבָר וְעִם בְּנֵי אָדָם שֶׁגָּרְמוּ לָנוּ אַכְזָבָה, סֵבֶל וְיִסּוּרִים — מִשְׁקָעִים אֵלּוּ הֵם הַמְּנַהֲלִים אוֹתָנוּ מִבִּפְנִים: אֶת מַצַּב רוּחֵנוּ וְנַפְשֵׁנוּ, בְּרִיאוּת גּוּפֵנוּ, וְאֶת אֹפֶן הִתְמוֹדְדוּתֵנוּ עִם אֶתְגְּרֵי הַחַיִּים.
שְׁלֹשָׁה שְׁלָבִים שֶׁל הַכְנָעָה, לִפְנֵי הַכֹּל
בְּכָל אֶחָד מִשְּׁלֹשֶׁת מַסְלוּלֵי הַפִּיּוּס פּוֹתְחִים תָּמִיד בְּאוֹתָם שְׁלֹשָׁה שְׁלָבִים: לְהוֹדוֹת בַּפְּגִיעָה בְּלִי לְהַכְחִישׁ וּבְלִי לְצַמְצֵם, לְבַטֵּא אֶת הַכְּאֵב בְּקוֹל, וְלְהִתְמוֹדֵד עִמּוֹ עַד שֶׁעוֹבְרִים אֶל עֵבֶר הַשֵּׁנִי שֶׁלּוֹ. רַק אַחֲרֵי זֶה אֶפְשָׁר לְהִבָּדֵל מִן הַמַּשְׁקָע וּלְהַתְחִיל לְתַקֵּן אוֹתוֹ.
שְׁלֹשָׁה מַסְלוּלֵי פִּיּוּס
"קָרְבָה אֶל נַפְשִׁי גְאָלָהּ" (תהלים סט יט) — גְּאֻלָּה פְּרָטִית שֶׁל הַנֶּפֶשׁ מִגָּלוּת הַיֵּצֶר הָרַע. "וּכְשֶׁיִּגְאַל כָּל אֶחָד גְּאֻלָּה פְּרָטִית — אָז יִהְיֶה אַחַר כָּךְ גְּאֻלָּה כְּלָלִית". (בעל שם טוב על התורה, בראשית קסו)
פִּיּוּס עִם הַקְּרוֹבִים
עוֹמְדִים מוּל הַפּוֹגֵעַ
בְּקַלְסְתֵּר פָּנָיו הַמָּלֵא, כְּפִי שֶׁהוּא מְצֻיָּר בְּתוֹךְ הַנֶּפֶשׁ בִּשְׁעַת הַפְּגִיעָה. זוֹ לֹא עֲבוֹדָה שֶׁל דִּמְיוֹן — זוֹ תְּמוּנָה פְּנִימִית אֲמִתִּית שֶׁמְּפַעֶלֶת אוֹתָנוּ, וְאִתָּהּ צָרִיךְ לְהִתְמוֹדֵד.
רוֹשְׁמִים אֶת כָּל מִי שֶׁפָּגַע בָּנוּ, אִכְזֵב אוֹתָנוּ וְשֶׁאָנוּ כּוֹעֲסִים עָלָיו. נוֹתְנִים מָקוֹם וְכָבוֹד לַחֲוָיָה, בְּחֹמֶר חֲוָיָתִי מַמָּשִׁי וְלֹא בְּאִידֵאָל שֶׁלֹּא קַיָּם אֶצְלֵנוּ.
מַקְדִּישִׁים לְמַשִּׂימַת הַפִּיּוּס לְפָחוֹת בֹּקֶר אֶחָד שָׁלֵם, מֻקְדָּם כְּכָל הָאֶפְשָׁר.
מְחַפְּשִׂים מָקוֹם שֶׁאֶפְשָׁר לְהִתְבַּטֵּא בּוֹ בְּלִי בּוּשָׁה — חֶדֶר, הַר גָּבוֹהַּ, שָׂדֶה פָּתוּחַ, יַעַר אוֹ חוֹף הַיָּם. אָסוּר לְהִתְבּוֹדֵד לְבַד — תָּמִיד עִם מְלַוֶּה שֶׁמַּכִּיר אֶת הַתַּהֲלִיךְ.
גַּם בְּתוֹךְ הַמָּקוֹם שֶׁבָּחַרְנוּ מְחַפְּשִׂים אֶת הַפִּנָּה הַמְּדֻיֶּקֶת — "כָּמוֹ אֲרֻבָּה" — שֶׁבָּהּ מַצְלִיחִים לְהִתְבַּטֵּא יוֹתֵר.
בּוֹחֲרִים מִן הָרְשִׁימָה אֶת הַפְּגִיעָה הַמְּכוֹנֶנֶת, הַקָּשָׁה בְּיוֹתֵר, וְעוֹמְדִים מוּל הַפּוֹגֵעַ מֵהָרֹאשׁ וְעַד הָרַגְלַיִם, כְּפִי שֶׁהוּא נִזְכָּר אֶצְלֵנוּ בִּשְׁעַת הַפְּגִיעָה.
מְיַשְּׁרִים מַבָּט וּמַתְחִילִים לִצְעֹק, בְּלֹא סִנּוּן שִׂכְלִי. פָּשׁוּט מוֹצִיאִים מַה שֶּׁיּוֹצֵא, כְּמוֹ הַר גַּעַשׁ. עַל מַה שֶּׁהִשְׁתַּקֵּעַ בְּתוֹכֵנוּ מֻתָּר לִצְעֹק — גַּם כְּשֶׁהַפּוֹגֵעַ הוּא אַבָּא אוֹ אִמָּא.
מוֹנִים אֶחָד לְאֶחָד, בְּקֹר וּבְסֵדֶר, אֶת כָּל הָעֲוָלוֹת — גַּם מַה שֶּׁצִּפִּינוּ וְלֹא קִבַּלְנוּ.
לֹא מְבַקְּשִׁים — דּוֹרְשִׁים. נוֹתְנִים פְּקֻדּוֹת בְּרוּרוֹת מַה צָּרִיךְ לְתַקֵּן, אֶחָד אַחֲרֵי הַשֵּׁנִי, לְפִי הָרְשִׁימָה.
רוֹאִים בְּעֵינֵי הָרוּחַ אֵיךְ הַפּוֹגֵעַ בָּא וּמְתַקֵּן בְּפֹעַל אֶת מַה שֶּׁקִּלְקֵל, מַמָּשׁ כְּמוֹ בְּאוֹתוֹ זְמַן שֶׁהָיְתָה הַפְּגִיעָה, רַק הָפוּךְ.
מְסָרְבִים לְקַבֵּל בַּהַתְחָלָה, רוֹאִים אוֹתוֹ מֻבָּס וּמֻשְׁפָּל. זוֹ נְקֻדָּה הֶכְרֵחִית לְרִפּוּי הַנֶּפֶשׁ וּלְהָשָׁבַת הַצֶּדֶק — לֹא נְקָמָה בַּמְּצִיאוּת, אֶלָּא בַּתְּמוּנָה הַפְּנִימִית שֶׁמְּנַהֶלֶת אוֹתָנוּ.
חוֹזְרִים לְמִדָּתִיּוּת וּמַזְכִּירִים גַּם אֶת הַדְּבָרִים הַטּוֹבִים שֶׁהַפּוֹגֵעַ כֵּן עָשָׂה אִתָּנוּ.
אוֹמְרִים תּוֹדָה עַל הַדְּבָרִים הַטּוֹבִים. לִפְעָמִים זֶה הַשָּׁלָב הַקָּשֶׁה בְּיוֹתֵר.
קוֹבְעִים כְּלָלִים חֲדָשִׁים וּבְרוּרִים לַיַּחֲסִים מִכָּאן וָהָלְאָה. "אִם אַתָּה כָּכָה כְּלַפַּי, אֲנִי כָּכָה כְּלַפֶּיךָ."
מִשְׁתַּחְרְרִים מֵהַמִּטְעָן שֶׁל הַזַּעַם, הָעֶלְבּוֹן וְהַחֻלְשָׁה. אֲנַחְנוּ כְּבָר שׁוֹלְטִים בַּמַּצָּב, וְלָכֵן יְכוֹלִים לְהִתְפַּיֵּס.
פּוֹנִים בְּקוֹל רָם, בָּרוּר וְשָׁקֵט, וְרוֹאִים אֶת הַפּוֹגֵעַ פּוֹנֶה אֵלֵינוּ בְּכָבוֹד וּבְאַהֲבָה. דַּף חָדָשׁ, מִתּוֹךְ עֶמְדַּת כֹּחַ.
חָשׁוּב מְאֹד
אָסוּר לְהַפְסִיק בְּאֶמְצַע הַתַּהֲלִיךְ. כָּל הַשְּׁלָבִים הוֹלְכִים בְּרֶצֶף אֶחָד עַד הַסּוֹף — בְּדִיּוּק כְּמוֹ נִתּוּחַ שֶׁאָסוּר לְהַשְׁאִיר פָּתוּחַ בְּאֶמְצָעוֹ. אִם נִשְׁאָר מַשֶּׁהוּ, חוֹזְרִים עַל הַתִּקּוּן לְמָחֳרָת בַּבֹּקֶר.
פִּיּוּס עִם ה'
עֲמִידָה לִפְנֵי מִי שֶׁאֵין לוֹ גּוּף
הוּא מְמַלֵּא אֶת כָּל הָאָרֶץ, בְּלִי שׁוּלַיִם וּבְלִי גְּבוּלוֹת. עוֹמְדִים מוּלוֹ וּמוֹצִיאִים הַחוּצָה אֶת מַה שֶּׁכּוֹאֵב — הוֹצָאַת הַמֻּגְלָה וְהַחֳלִי.
עַל מָה אֲנִי כּוֹעֵס? מַה חָסֵר לִי וּמַה מְּיֻתָּר אֶצְלִי? זֶה אִבְחוּן מְקוֹם הַכְּאֵב וְהַחֳלִי, לִפְנֵי שֶׁמּוֹצִיאִים אוֹתוֹ.
הַנְּכוֹנוּת לָגֶשֶׁת לַמְּשִׂימָה בַּמָּקוֹם שֶׁבָּחַרְנוּ, וּלְדַבֵּר אִתּוֹ בְּרֶצֶף אֶחָד, מֵהַהַתְחָלָה וְעַד הַסּוֹף, בְּלִי לַעֲצֹר.
מוֹצִיאִים אֶת כָּל הַ"לָּמָּה" — לָמָּה כָּכָה, לָמָּה חָסֵר לִי זֶה, לָמָּה הֵבֵאתָ לִי אֶת הַנִּסָּיוֹן הַזֶּה. כְּמוֹ מֹשֶׁה רַבֵּנוּ: "לָמָּה הֲרֵעֹתָ".
מְבַקְּשִׁים מִמֶּנּוּ יִתְבָּרֵךְ שְׁמוֹ לְמַלֵּא אֶת חֶסְרוֹנֵנוּ וּלְהָסִיר אֶת הַמְּצוּקָה. הוּא כֹּל יָכוֹל.
מְנַקִּים עַד הַסּוֹף, לֹא מַשְׁאִירִים כְּלוּם בִּפְנִים — אַל תַּגִּיד "זֶה מַסְפִּיק", תּוֹצִיא וְתוֹצִיא. כְּמוֹ מְנַתֵּחַ שֶׁחוֹזֵר לִסְגֹּר בְּדִיּוּק אֶת מַה שֶּׁפָּתַח.
מַזְכִּירִים בְּקוֹל רָם אֶת הַדְּבָרִים הַטּוֹבִים שֶׁעָשָׂה אִתָּנוּ, וְאוֹמְרִים תּוֹדָה. הַכָּרַת הַטּוֹב נוֹתֶנֶת כֹּחַ לְהִתְאוֹשֵׁשׁ.
קוֹבְעִים בְּרִית מְחֻדֶּשֶׁת — כְּלָלִים חֲדָשִׁים שֶׁל יַחֲסִים, מַה שֶּׁאֲנִי צָרִיךְ מִמֶּנּוּ מִכָּאן וָהָלְאָה, וְגַם הִתְחַיְּבוּת שֶׁלִּי כְּלַפָּיו.
מִתְרוֹקְנִים מֵהָעֶלְבּוֹן וְהָאַכְזָבָה, וּמְמַלְּאִים אֶת הַמָּקוֹם הַפָּנוּי בְּכֹחוֹת חֲדָשִׁים שֶׁל שִׂמְחָה, חֶמְלָה וְהוֹדָיָה. זוֹ הַחֲלָמָה — חִבּוּר מֵחָדָשׁ אֶל הָאֲנִי הַמְּקוֹרִי בֶּן הַחוֹרִין.
רוֹאִים אוֹתוֹ פּוֹנֶה אֵלֵינוּ בְּכָבוֹד וּבְחִבָּה יְתֵרָה, וּמַתְחִילִים לְהַרְגִּישׁ אֶת הַהַשְׁגָּחָה שֶׁל רַחֲמִים. מִזְדַּהִים עִם הַמַּצָּב הַנַּפְשִׁי הֶחָדָשׁ.
כָּל שֶׁדָּבָר יוֹתֵר אֲמִתִּי, הוּא יוֹתֵר פָּשׁוּט. "בְּבוֹא הַשִּׂמְחָה וְהָאַהֲבָה יִסְתַּלֵּק הַכַּעַס" — הַדִּינִים עוֹלִים אֶל שָׁרְשָׁם וְשָׁם הֵם נִמְתָּקִים.
פָּסוּק מוּמְלָץ
"קָרְבָה אֶל נַפְשִׁי גְאָלָהּ" — תהלים סט יט
פִּיּוּס עִם עַצְמִי
פִּנּוּי מָקוֹם לַשֶּׁפַע
עוֹמְדִים מוּל עַצְמֵנוּ כְּפִי שֶׁעָמַדְנוּ מוּל הַפּוֹגֵעַ וּמוּל הַשֵּׁם. יֵשׁ שֶׁפַע מוּכָן לָאָדָם, אֲבָל צָרִיךְ קֹדֶם לְפַנּוֹת מָקוֹם חָדָשׁ — מִכָּל הַמַּשְׁקָעִים, מִכָּל הַפַּאשְׁלוֹת וְהַחֶסְרוֹנוֹת.
עַל מָה בְּדִיּוּק אֲנִי כּוֹעֵס עַל עַצְמִי? מְזַהִּים בְּלִי לְהִתְעַכֵּב — כְּדֵי לֹא לְהַקְפִּיד עַל עַצְמֵנוּ.
פּוֹתְחִים אֶת הַדְּבָרִים בְּלֵב, מוּל עַצְמֵנוּ, בְּרֶצֶף אֶחָד מֵהַהַתְחָלָה וְעַד הַסּוֹף.
עוֹמְדִים מוּל עַצְמֵנוּ וּמוֹצִיאִים אֶת כָּל הָעֲוָלוֹת שֶׁעָשִׂינוּ לְעַצְמֵנוּ וְלַאֲחֵרִים — בְּלִי לְקַלֵּל אֶת עַצְמֵנוּ, רַק לְהוֹצִיא וּלְהוֹדוֹת.
דּוֹרְשִׁים מֵעַצְמֵנוּ לְתַקֵּן כָּעֵת וּמִיָּד אֶת מַה שֶּׁאֶפְשָׁר.
רוֹאִים בְּעֵינֵי הָרוּחַ אֵיךְ אֲנַחְנוּ מְבַצְּעִים אֶת הַתִּקּוּן בְּפֹעַל, מַמָּשׁ בַּמְּצִיאוּת. זֶה בְּעַצְמוֹ כְּבָר סְגִירָה שֶׁל הַמָּקוֹם הַפָּגוּעַ.
מְכִינִים תָּכְנִית בְּרוּרָה: אֵיךְ פּוֹעֲלִים וּמָה מַשְׁלִימִים מִמַּה שֶּׁחָסֵר. בֵּינִי לְבֵין עַצְמִי בִּלְבַד.
מַזְכִּירִים לְעַצְמֵנוּ בְּקוֹל רָם אֶת הַדְּבָרִים הַטּוֹבִים שֶׁעָשִׂינוּ. חֶדֶר הַהִתְאוֹשְׁשׁוּת — "יִישַׁר כֹּחַ, יֵשׁ לְךָ נְשָׁמָה טוֹבָה".
קוֹבְעִים הַנְחָיָה בְּרוּרָה לְכָל כֹּחוֹת הַנֶּפֶשׁ, כְּדֵי שֶׁהָעֶגְלוֹן יְנַהֵג — לֹא הַסּוּסִים.
מִתְרוֹקְנִים מֵהָאַכְזָבָה מֵעַצְמֵנוּ, וּמְמַלְּאִים אֶת הַמָּקוֹם בְּכֹחוֹת חֲדָשִׁים שֶׁל שִׂמְחָה שֶׁל הַתְחָלָה חֲדָשָׁה.
פּוֹתְחִים שְׁבִיל חָדָשׁ בַּחַיִּים, וְרוֹאִים אֶת הַנְּשָׁמָה מְאִירָה בְּתוֹכֵנוּ — בָּרוּחַ, בַּנֶּפֶשׁ וּבְכָל כֹּחוֹתֵינוּ.
"כִּי לְכָל הָעָם בִּשְׁגָגָה" — אֲנַחְנוּ לוֹמְדִים יוֹתֵר מֵהַטָּעֻיּוֹת שֶׁלָּנוּ מֵאֲשֶׁר מֵהַהַצְלָחוֹת. הַהַצְלָחוֹת הֵן מֵהַשֵּׁם יִתְבָּרֵךְ, וְאֶת הַפַּאשְׁלוֹת שֶׁלָּנוּ אָנוּ לוֹמְדִים לְתַקֵּן.
